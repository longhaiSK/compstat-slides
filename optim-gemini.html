This interactive animation demonstrates how different optimization algorithms navigate bivariate surfaces.

  • Orange Arrow: The gradient \(\nabla f\) at the current point (normalized for visibility). It points in the direction of steepest ascent.
  • Green Arrow: The actual step \(p_k\) taken by the algorithm.

# --- Objective Functions Definition ---
fns <- list(
  "Good Quadratic" = list(
    f = function(x) x[1]^2 + x[2]^2,
    g = function(x) c(2*x[1], 2*x[2]),
    H = function(x) matrix(c(2,0,0,2), 2, 2),
    xlim = c(-2, 2), ylim = c(-2, 2),
    default_start = c(-1.8, -1.2)
  ),
  "Narrow Ridge (Rosenbrock)" = list(
    f = function(x) (1-x[1])^2 + 100*(x[2]-x[1]^2)^2,
    g = function(x) c(-2*(1-x[1]) - 400*x[1]*(x[2]-x[1]^2), 200*(x[2]-x[1]^2)),
    H = function(x) matrix(c(2-400*x[2]+1200*x[1]^2, -400*x[1], -400*x[1], 200), 2, 2),
    xlim = c(-2, 2), ylim = c(-1, 3),
    default_start = c(-1.5, 2.5)
  ),
  "Multi-Mode (Himmelblau)" = list(
    f = function(x) (x[1]^2+x[2]-11)^2 + (x[1]+x[2]^2-7)^2,
    g = function(x) c(4*x[1]*(x[1]^2+x[2]-11) + 2*(x[1]+x[2]^2-7), 
                      2*(x[1]^2+x[2]-11) + 4*x[2]*(x[1]+x[2]^2-7)),
    H = function(x) matrix(c(12*x[1]^2 + 4*x[2] - 42, 4*x[1] + 4*x[2],
                             4*x[1] + 4*x[2], 4*x[1] + 12*x[2]^2 - 26), 2, 2),
    xlim = c(-5, 5), ylim = c(-5, 5),
    default_start = c(-4, -4)
  )
)

# --- Backtracking Line Search ---
line_search <- function(f, g, x, p, max_iter=20) {
  a <- 1
  c1 <- 1e-4
  fx <- f(x)
  gx <- g(x)
  for(i in 1:max_iter) {
    if(f(x + a*p) <= fx + c1*a*sum(gx*p)) break
    a <- a * 0.5
  }
  return(a)
}

# --- Core Optimization Engine ---
run_optim <- function(method, fn, x0, max_iter=50) {
  f <- fn$f; g <- fn$g; H <- fn$H
  x <- x0
  path <- data.frame(iter=1, x=x[1], y=x[2], gx=NA, gy=NA, dx=NA, dy=NA)
  
  B <- diag(2)
  p_old <- rep(0, 2)
  gx_old <- rep(0, 2)
  
  for(k in 1:max_iter) {
    gx <- g(x)
    if(sqrt(sum(gx^2)) < 1e-4) break # Converged
    
    # Calculate search direction
    if(method == "GD + Line Search") {
      pk <- -gx
    } else if(method == "Newton-Raphson") {
      Hk <- H(x)
      # Ensure positive definiteness to force descent direction
      eig <- eigen(Hk)$values
      if(min(eig) < 1e-5) Hk <- Hk + diag(2) * (abs(min(eig)) + 1e-2)
      pk <- -solve(Hk, gx)
    } else if(method == "BFGS") {
      pk <- - as.vector(B %*% gx)
    } else if(method == "Conjugate Gradient") {
      if(k == 1) {
        pk <- -gx
      } else {
        beta <- sum(gx^2) / sum(gx_old^2) # Fletcher-Reeves
        pk <- -gx + beta * p_old
      }
    }
    
    # Step size
    alpha <- line_search(f, g, x, pk)
    x_next <- x + alpha * pk
    
    # BFGS Update
    if(method == "BFGS") {
      sk <- x_next - x
      yk <- g(x_next) - gx
      rho <- 1/sum(yk*sk)
      if(rho > 0 && !is.infinite(rho)) {
        I <- diag(2)
        B <- (I - rho * sk %*% t(yk)) %*% B %*% (I - rho * yk %*% t(sk)) + rho * sk %*% t(sk)
      }
    }
    
    # Save path data
    path$gx[k] <- gx[1]
    path$gy[k] <- gx[2]
    path$dx[k] <- x_next[1] - x[1]
    path$dy[k] <- x_next[2] - x[2]
    
    x_old <- x; gx_old <- gx; p_old <- pk
    x <- x_next
    
    path <- rbind(path, data.frame(iter=k+1, x=x[1], y=x[2], gx=NA, gy=NA, dx=NA, dy=NA))
  }
  return(path)
}

# --- Shiny UI ---
ui <- page_sidebar(
  title = "Bivariate Optimization Animation",
  sidebar = sidebar(
    selectInput("func", "Objective Function:", choices = names(fns)),
    selectInput("algo", "Algorithm:", choices = c("GD + Line Search", "Newton-Raphson", "Conjugate Gradient", "BFGS")),
    fluidRow(
      column(6, numericInput("start_x", "Start X:", value = -1.8, step = 0.5)),
      column(6, numericInput("start_y", "Start Y:", value = -1.2, step = 0.5))
    ),
    actionButton("run", "Compute Path", class = "btn-primary", width = "100%"),
    hr(),
    uiOutput("slider_ui"),
    p("Click the Play button above to animate the steps.", style = "font-size: 0.85em; color: gray;")
  ),
  card(
    plotOutput("optim_plot", height = "600px")
  )
)

# --- Shiny Server ---
server <- function(input, output, session) {
  
  # Update default start coordinates when function changes
  observeEvent(input$func, {
    req(input$func)
    defaults <- fns[[input$func]]$default_start
    updateNumericInput(session, "start_x", value = defaults[1])
    updateNumericInput(session, "start_y", value = defaults[2])
  })
  
  # Reactive optimization calculation
  optim_data <- eventReactive(input$run, {
    fn <- fns[[input$func]]
    run_optim(input$algo, fn, c(input$start_x, input$start_y))
  }, ignoreNULL = FALSE)
  
  # Render dynamic animation slider
  output$slider_ui <- renderUI({
    path <- optim_data()
    max_steps <- max(1, nrow(path) - 1)
    sliderInput("step", "Animation Step:", 
                min = 1, max = max_steps, value = 1, step = 1,
                animate = animationOptions(interval = 800, loop = FALSE))
  })
  
  # Background contour data caching
  contour_grid <- reactive({
    fn <- fns[[input$func]]
    x_seq <- seq(fn$xlim[1], fn$xlim[2], length.out = 100)
    y_seq <- seq(fn$ylim[1], fn$ylim[2], length.out = 100)
    grid <- expand.grid(x = x_seq, y = y_seq)
    grid$z <- apply(grid, 1, fn$f)
    # Log scale z for better visual contours, especially on Rosenbrock
    grid$z <- log10(grid$z - min(grid$z) + 1e-4) 
    return(list(grid=grid, fn=fn))
  })
  
  # Main Plot
  output$optim_plot <- renderPlot({
    req(input$step)
    path <- optim_data()
    cg <- contour_grid()
    
    current_step <- input$step
    path_to_current <- path[1:current_step, ]
    current_pt <- path[current_step, ]
    
    p <- ggplot() +
      geom_contour(data = cg$grid, aes(x=x, y=y, z=z), bins = 25, color = "grey70") +
      geom_path(data = path_to_current, aes(x=x, y=y), color = "black", linewidth = 0.8) +
      geom_point(data = path_to_current, aes(x=x, y=y), size = 2) +
      coord_cartesian(xlim = cg$fn$xlim, ylim = cg$fn$ylim) +
      theme_minimal() +
      labs(x = "X", y = "Y", title = paste(input$algo, "on", input$func))
    
    # If not the last step, draw arrows
    if(!is.na(current_pt$gx) && !is.na(current_pt$dx)) {
      
      # 1. Normalize gradient vector length for clear visualization
      g_mag <- sqrt(current_pt$gx^2 + current_pt$gy^2)
      arrow_len <- (cg$fn$xlim[2] - cg$fn$xlim[1]) * 0.1 # Length = 10% of plot width
      scaled_gx <- (current_pt$gx / g_mag) * arrow_len
      scaled_gy <- (current_pt$gy / g_mag) * arrow_len
      
      # 2. Draw Gradient (Orange)
      p <- p + geom_segment(
        aes(x = current_pt$x, y = current_pt$y, 
            xend = current_pt$x + scaled_gx, yend = current_pt$y + scaled_gy),
        color = "orange", arrow = arrow(length = unit(0.3, "cm"), type="closed"), 
        linewidth = 1.2
      )
      
      # 3. Draw Actual Jump Step (Green)
      p <- p + geom_segment(
        aes(x = current_pt$x, y = current_pt$y, 
            xend = current_pt$x + current_pt$dx, yend = current_pt$y + current_pt$dy),
        color = "green", arrow = arrow(length = unit(0.3, "cm"), type="closed"), 
        linewidth = 1.2
      )
      
      # Add Legend Annotations manually for clarity
      p <- p + annotate("text", x = cg$fn$xlim[1], y = cg$fn$ylim[2], 
                        label = "Orange: Gradient (Direction of Steepest Ascent)\nGreen: Actual Optimization Step", 
                        hjust = 0, vjust = 1, color = "black", fontface = "bold", size = 4)
    }
    
    return(p)
  })
}

shinyApp(ui, server)
